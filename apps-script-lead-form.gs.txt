function doPost(e) {
  var d = e.parameter || {};
  if (d.website) { return ContentService.createTextOutput('ok'); } // honeypot
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var now = new Date();
  sheet.appendRow([now, d.name || '', d.phone || '', d.address || '', d.interest || '', d.message || '']);
  MailApp.sendEmail('barrettjessen21@gmail.com', 'New gutter lead: ' + (d.name || 'Unknown') + ' - Houston Gutter Cleaning',
    'Name: ' + (d.name || '') + '\nPhone: ' + (d.phone || '') + '\nAddress/ZIP: ' + (d.address || '') +
    '\nService: ' + (d.interest || '') + '\nMessage: ' + (d.message || '') + '\nSubmitted: ' + now);
  return ContentService.createTextOutput(JSON.stringify({result: 'success'})).setMimeType(ContentService.MimeType.JSON);
}
